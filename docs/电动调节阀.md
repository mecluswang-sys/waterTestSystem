### 3.1 模拟量转换

```cpp
inline int percentToAO(float percent) {
    if (percent < 0) percent = 0;
    if (percent > 100) percent = 100;
    return static_cast<int>(percent / 100.0 * 22118 + 5530);
}

inline float aiToPercent(int value) {
    return static_cast<float>(value) / 27648.0f * 100.0f;
}
```

### 3.3 IO管理

```cpp
class IOManager {
public:
    IOManager(ModbusClient& client) : client_(client) {}

    void setValvePercent(float percent) {
        int ao = percentToAO(percent);
        client_.writeRegister(40001, ao);
    }

    float getPressure() {
        int val = client_.readRegister(30001);
        return aiToPercent(val);
    }

private:
    ModbusClient& client_;
};
```

### 3.4 PID（建议加限幅）

```cpp
class PID {
public:
    PID(double kp, double ki, double kd)
        : kp_(kp), ki_(ki), kd_(kd) {}

    double compute(double setpoint, double actual) {
        double error = setpoint - actual;

        integral_ += error;
        if (integral_ > 1000) integral_ = 1000;
        if (integral_ < -1000) integral_ = -1000;

        double derivative = error - prev_error_;
        prev_error_ = error;

        double output = kp_ * error + ki_ * integral_ + kd_ * derivative;

        // 输出限幅（0~100%）
        if (output > 100) output = 100;
        if (output < 0) output = 0;

        return output;
    }

private:
    double kp_, ki_, kd_;
    double prev_error_ = 0;
    double integral_ = 0;
};
```

---

### 3.5 阀门控制器

```cpp
class ValveController {
public:
    ValveController(IOManager& io, PID& pid)
        : io_(io), pid_(pid) {}

    void update(double target_pressure) {
        double pressure = io_.getPressure();

        double valve = pid_.compute(target_pressure, pressure);

        io_.setValvePercent(valve);
    }

private:
    IOManager& io_;
    PID& pid_;
};
```

## 4. 主循环（核心）

```cpp
#include <thread>
#include <chrono>

int main() {
    ModbusClient client;
    IOManager io(client);
    PID pid(0.5, 0.01, 0.1);

    ValveController controller(io, pid);

    double target_pressure = 500.0;

    while (true) {
        controller.update(target_pressure);

        std::this_thread::sleep_for(
            std::chrono::milliseconds(100)
        );
    }

    return 0;
}
```

## 5. 调试建议（非常关键）

### ✔ 第一步：开环测试

```cpp
io.setValvePercent(20);
```

确认：

* 阀门是否动作
* 开度是否变化

---

### ✔ 第二步：压力读取

```cpp
float p = io.getPressure();
```

确认：

* 数值是否稳定
* 是否有噪声

---

### ✔ 第三步：闭环控制

* 从小 KP 开始（如 0.2）
* KI 慢慢加（防震荡）
